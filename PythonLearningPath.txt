def fibonacci(n):
    a, b = 0, 1
    for _ in range(n):
        yield a
        a, b = b, a + b

    print(num)

🔹 Mini Project for Day 3
Task Manager App

Add tasks with deadlines.

Use decorators for logging.

Use context managers for file saving.

Try multithreading for reminders.
import json
import threading
import time
from datetime import datetime

# -------------------------------
# Decorator Example: Logging
# -------------------------------
def logger(func):
    def wrapper(*args, **kwargs):
        print(f"[LOG] Calling {func.__name__} at {datetime.now()}")
        return func(*args, **kwargs)
    return wrapper

# -------------------------------
# Task Class
# -------------------------------
class Task:
    def __init__(self, title, deadline):
        self.title = title
        self.deadline = deadline

    def __str__(self):
        return f"Task: {self.title}, Deadline: {self.deadline}"

# -------------------------------
# Task Manager Class
# -------------------------------
class TaskManager:
    def __init__(self, filename="tasks.json"):
        self.filename = filename
        self.tasks = []
        self.load_tasks()

    @logger
    def add_task(self, title, deadline):
        task = Task(title, deadline)
        self.tasks.append(task)
        self.save_tasks()
        print("Task added successfully!")

    def view_tasks(self):
        if not self.tasks:
            print("No tasks available.")
            return
        for task in self.tasks:
            print(task)

    def save_tasks(self):
        with open(self.filename, "w") as f:
            json.dump([t.__dict__ for t in self.tasks], f)

    def load_tasks(self):
        try:
            with open(self.filename, "r") as f:
                data = json.load(f)
                self.tasks = [Task(**t) for t in data]
        except FileNotFoundError:
            self.tasks = []

# -------------------------------
# Context Manager Example
# -------------------------------
class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode

    def __enter__(self):
        self.file = open(self.filename, self.mode)
        return self.file

    def __exit__(self, exc_type, exc_val, exc_tb):
        self.file.close()

# -------------------------------
# Multithreading Example: Reminder
# -------------------------------
def reminder(task):
    while True:
        now = datetime.now().strftime("%Y-%m-%d %H:%M")
        if now == task.deadline:
            print(f"Reminder: Task '{task.title}' is due now!")
            break
        time.sleep(30)  # check every 30 seconds

# -------------------------------
# CLI Menu
# -------------------------------
if __name__ == "__main__":
    manager = TaskManager()

    while True:
        print("\nTask Manager")
        print("1. Add Task")
        print("2. View Tasks")
        print("3. Exit")

        choice = input("Enter your choice (1-3): ")

        if choice == '1':
            title = input("Enter task title: ")
            deadline = input("Enter deadline (YYYY-MM-DD HH:MM): ")
            manager.add_task(title, deadline)

            # Start reminder thread
            task = manager.tasks[-1]
            threading.Thread(target=reminder, args=(task,), daemon=True).start()

        elif choice == '2':
            manager.view_tasks()

        elif choice == '3':
            print("Exiting Task Manager. Goodbye!")
            break

        else:
            print("Invalid choice. Please try again.")

You can have multiple parameters:
